//@version=5
// LATEST | v1 EMA 9/20 + Volume + VWAP + RSI + ADX | INTRADAY ONLY
// Overlay. BUY CE / BUY PE on NIFTY index. Place at next candle open.
// No carryforward: new entries stop before square-off; EXIT ALL at square-off time.
indicator("v1 | EMA VWAP RSI ADX Intraday", shorttitle = "v1 EMA Intraday", overlay=true, max_labels_count=500)

// ================= INPUTS =================
fastLen     = input.int(9,   "Fast EMA", minval=1)
slowLen     = input.int(20,  "Slow EMA", minval=1)
trendLen    = input.int(50,  "Trend EMA (filter)", minval=1)
useTrend    = input.bool(true, "Use Trend EMA filter")

volLen      = input.int(20,  "Volume MA length", minval=1)
volMult     = input.float(1.3, "Volume multiplier (x avg)", minval=0.5, step=0.1)

useVwap     = input.bool(true, "Use VWAP filter")
useRsi      = input.bool(true, "Use RSI filter")
rsiLen      = input.int(14,  "RSI length")
rsiBuyMin   = input.int(55,  "RSI min for BUY CE")
rsiSellMax  = input.int(45,  "RSI max for BUY PE")

useAdx      = input.bool(true, "Use ADX (trend strength) filter")
adxMin      = input.int(20,  "ADX minimum")

string G_DAY = "Intraday (no carryforward)"
useSession  = input.bool(true, "Enforce intraday session", group = G_DAY)
entrySess   = input.session("0915-1445", "New entries only (IST)", group = G_DAY, tooltip = "No new BUY CE / BUY PE after this window. Avoids overnight holds.")
sqOffSess   = input.session("1515-1516", "Square-off alert (IST)", group = G_DAY, tooltip = "Shows EXIT ALL once per day. Close any open option before this.")
showSqOff   = input.bool(true, "Show EXIT ALL at square-off", group = G_DAY)

atrLen      = input.int(14,  "ATR length")
slMult      = input.float(1.5, "Stop-loss (x ATR)", step=0.1)
rrRatio     = input.float(2.0, "Target (Risk:Reward)", step=0.1)
showSLTP    = input.bool(true, "Show SL / Target in label")

confirmClose = input.bool(true, "Signal only on candle close", tooltip = "Stops labels from appearing mid-candle and disappearing.")

// ================= CALCULATIONS =================
emaFast  = ta.ema(close, fastLen)
emaSlow  = ta.ema(close, slowLen)
emaTrend = ta.ema(close, trendLen)

volAvg   = ta.sma(volume, volLen)
volOk    = volume > volAvg * volMult

vwapVal  = ta.vwap(hlc3)
rsiVal   = ta.rsi(close, rsiLen)
[diPlus, diMinus, adxVal] = ta.dmi(14, 14)
atrVal   = ta.atr(atrLen)

// Entry window only — after last-entry time, no new calls (no carryforward)
canEnter  = not useSession or not na(time(timeframe.period, entrySess, "Asia/Kolkata"))
atSqOff   = not na(time(timeframe.period, sqOffSess, "Asia/Kolkata"))

// ================= SIGNAL LOGIC =================
crossUp   = ta.crossover(emaFast, emaSlow)
crossDown = ta.crossunder(emaFast, emaSlow)

buyCond = crossUp and volOk and canEnter
buyCond := buyCond and (not useVwap  or close > vwapVal)
buyCond := buyCond and (not useRsi   or rsiVal >= rsiBuyMin)
buyCond := buyCond and (not useAdx   or adxVal >= adxMin)
buyCond := buyCond and (not useTrend or close > emaTrend)

sellCond = crossDown and volOk and canEnter
sellCond := sellCond and (not useVwap  or close < vwapVal)
sellCond := sellCond and (not useRsi   or rsiVal <= rsiSellMax)
sellCond := sellCond and (not useAdx   or adxVal >= adxMin)
sellCond := sellCond and (not useTrend or close < emaTrend)

buyFire  = buyCond  and (not confirmClose or barstate.isconfirmed)
sellFire = sellCond and (not confirmClose or barstate.isconfirmed)

// Square-off once per calendar day in Asia/Kolkata
var int lastSqOffDay = na
int  dayNow = year(time, "Asia/Kolkata") * 10000 + month(time, "Asia/Kolkata") * 100 + dayofmonth(time, "Asia/Kolkata")
bool sqOffFire = showSqOff and useSession and atSqOff and barstate.isconfirmed and (na(lastSqOffDay) or lastSqOffDay != dayNow)
if sqOffFire
    lastSqOffDay := dayNow

bgcolor(useSession and not canEnter and not atSqOff ? color.new(color.orange, 92) : na, title = "No new entries (intraday)")

// ================= PLOTS =================
plot(emaFast,  "EMA 9",  color=color.new(color.aqua, 0),   linewidth=2)
plot(emaSlow,  "EMA 20", color=color.new(color.orange, 0), linewidth=2)
plot(emaTrend, "EMA 50", color=color.new(color.gray, 40),  linewidth=1)
plot(vwapVal,  "VWAP",   color=color.new(color.purple, 0), linewidth=1)

barcolor(volOk ? (close >= open ? color.new(color.green, 0) : color.new(color.red, 0)) : na)

// ================= LABELS =================
if buyFire
    sl  = close - atrVal * slMult
    tgt = close + (close - sl) * rrRatio
    txt = "BUY CE (intraday)\n" + (showSLTP ? "SL: " + str.tostring(sl, "#.##") + "\nTGT: " + str.tostring(tgt, "#.##") + "\n" : "") + "SQOFF by 15:15"
    label.new(bar_index, low, txt, style=label.style_label_up, color=color.new(color.green, 0), textcolor=color.white, size=size.small)

if sellFire
    sl  = close + atrVal * slMult
    tgt = close - (sl - close) * rrRatio
    txt = "BUY PE (intraday)\n" + (showSLTP ? "SL: " + str.tostring(sl, "#.##") + "\nTGT: " + str.tostring(tgt, "#.##") + "\n" : "") + "SQOFF by 15:15"
    label.new(bar_index, high, txt, style=label.style_label_down, color=color.new(color.red, 0), textcolor=color.white, size=size.small)

if sqOffFire
    label.new(bar_index, high, "EXIT ALL\nSquare off — no overnight", style=label.style_label_down, color=color.new(color.maroon, 0), textcolor=color.white, size=size.normal)
    alert("v1 EMA Intraday | EXIT ALL | Square off " + syminfo.ticker + " — no carryforward", alert.freq_once_per_bar_close)

// ================= ALERTS =================
alertcondition(buyFire,  "BUY CE (intraday)",  "v1 EMA Intraday: BUY CE on {{ticker}} @ {{close}} | square off same day")
alertcondition(sellFire, "BUY PE (intraday)",  "v1 EMA Intraday: BUY PE on {{ticker}} @ {{close}} | square off same day")
alertcondition(sqOffFire, "EXIT ALL square-off", "v1 EMA Intraday: EXIT ALL — square off, no overnight")
