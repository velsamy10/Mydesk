// Parse KEY=value from the NSE bridge paste line (…|KEY=value|…)
f_bridgeField(string src, string key) =>
    string needle = key + "="
    int    i      = str.pos(src, needle)
    string out    = ""
    if not na(i)
        string rest = str.substring(src, i + str.length(needle), str.length(src))
        int    pipe = str.pos(rest, "|")
        out := na(pipe) ? rest : str.substring(rest, 0, pipe)
    out

f_bridgeNum(string src, string key) =>
    string raw = f_bridgeField(src, key)
    raw == "" or str.lower(raw) == "n/a" ? na : str.tonumber(raw)

// 1 Long Buildup, 2 Short Buildup, 3 Short Covering, 4 Long Unwinding, 0 flat, na = no data
f_state(float pxCur, float pxRef, float oiCur, float oiRef, float minPct) =>
    float oiPct = not na(oiCur) and oiRef > 0 ? (oiCur - oiRef) / oiRef * 100.0 : na
    int   st    = na
    if not na(oiPct) and not na(pxCur) and not na(pxRef)
        bool oiUp = oiPct >= minPct
        bool oiDn = oiPct <= -minPct
        bool pxUp = pxCur > pxRef
        bool pxDn = pxCur < pxRef
        st := pxUp and oiUp ? 1 : pxDn and oiUp ? 2 : pxUp and oiDn ? 3 : pxDn and oiDn ? 4 : 0
    [st, oiPct]

f_isUp(int st) =>
    st == 1 or st == 3

f_isDn(int st) =>
    st == 2 or st == 4

f_oiText(int st) =>
    na(st) ? "Unavailable" : st == 1 ? "Long Buildup" : st == 2 ? "Short Buildup" : st == 3 ? "Short Covering" : st == 4 ? "Long Unwinding" : "Flat"

f_oiDirText(float pct) =>

// =====================================================================
// PCR = Put OI / Call OI across the complete CE/PE pairs you set
// =====================================================================
f_pcr(float c1, float p1, float c2, float p2, float c3, float p3) =>
    bool  k1 = c1 > 0 and not na(p1)
    bool  k2 = c2 > 0 and not na(p2)
    bool  k3 = c3 > 0 and not na(p3)
    float sc = (k1 ? c1 : 0.0) + (k2 ? c2 : 0.0) + (k3 ? c3 : 0.0)
    float sp = (k1 ? p1 : 0.0) + (k2 ? p2 : 0.0) + (k3 ? p3 : 0.0)
    int   n  = (k1 ? 1 : 0) + (k2 ? 1 : 0) + (k3 ? 1 : 0)
    float r  = sc > 0 ? sp / sc : na
    [r, n]

f_addLevel(float p, float vr, bool ob) =>
    bool  merged = false
    float tol    = atr * mergeAtr
    int   n      = array.size(levels)
    if n > 0
        for i = 0 to n - 1
            Level L = array.get(levels, i)
            if not merged and math.abs(L.price - p) <= tol
                L.price     := (L.price * L.pivots + p) / (L.pivots + 1)
                L.pivots    := L.pivots + 1
                L.touches   := L.touches + 1
                L.volRatio  := math.max(L.volRatio, vr)
                L.oiBuild   := L.oiBuild or ob
                L.lastTouch := bar_index
                merged      := true
    if not merged
        array.push(levels, Level.new(p, bar_index - pivRight, 1, 1, vr, ob, bar_index))
    merged

// 2 = strong, 1 = moderate
f_strength(Level L) =>
    int sc = 0
    sc += L.volRatio >= volHighMult ? 1 : 0
    sc += L.touches >= strongTouches ? 1 : 0
    sc += L.touches >= strongTouches * 2 ? 1 : 0
    sc += L.oiBuild ? 1 : 0
    sc >= 2 ? 2 : 1

float pivVolBase  = volAvg[pivRight]
float pivVolRatio = pivVolBase > 0 ? pivVol / pivVolBase : 0.0
bool  pivOiBuild  = not oiDaily and chOiPct[pivRight] >= oiMinPct
